    """Remove one named clause from the committed 06-05 migration (exact-string; counts must match)."""
    import sys
    MIG = "supabase/migrations/20260926120000_async_edge_definer_hardening.sql"

      "M2b": ("  WHERE (current_user_id = (SELECT auth.uid()) OR (SELECT auth.role()) = 'service_role');\n", "  ;\n", 1),
      # M3: the F-093 filters, in both reminder loops
      "M3": ("      AND e.deleted_at IS NULL\n      AND e.status = 'approved'\n", "", 2),
      # M4: the C4 REVOKE ALL on cron_job_status
      "M4": ("REVOKE ALL ON FUNCTION public.cron_job_status() FROM PUBLIC, anon, authenticated;\n", "", 1),
    }
    s = open(MIG).read()
    for key in sys.argv[1:]:
        old, new, n = MUT[key]
        c = s.count(old)
        if c != n:
            sys.exit(f"{key}: expected {n} match(es), found {c}")
        s = s.replace(old, new)
        print(f"{key}: removed {n} occurrence(s)")
    open(MIG, "w").write(s)
