// Start probe for plan 05-04 Task 2: spawn `next start` with a blank service
// key and placeholder public vars, capture output for up to 20 s, then kill.
import { spawn } from "node:child_process";

const cwd = "/Users/adyan/Documents/GitHub/Event-Radar";
const env = {
  ...process.env,
  SUPABASE_SERVICE_ROLE_KEY: "",
  NEXT_PUBLIC_SUPABASE_URL: "https://placeholder.supabase.co",
  NEXT_PUBLIC_SUPABASE_ANON_KEY: "placeholder-key",
};
const started = Date.now();
const child = spawn("npx", ["next", "start", "-p", "3100"], { cwd, env, detached: true });
let out = "";
child.stdout.on("data", (d) => (out += d));
child.stderr.on("data", (d) => (out += d));

let exited = false;
let exitInfo = null;
child.on("exit", (code, signal) => {
  exited = true;
  exitInfo = { code, signal, afterMs: Date.now() - started };
});

// While it runs, try one request so "refuses to serve" is measured, not assumed.
let probe = "not attempted";
setTimeout(async () => {
  if (exited) { probe = "skipped: process already exited"; return; }
  try {
    const res = await fetch("http://localhost:3100/api/health", { signal: AbortSignal.timeout(5000) });
    probe = `GET /api/health -> ${res.status} at ${Date.now() - started} ms`;
  } catch (e) {
    probe = `GET /api/health -> request failed: ${e.cause?.code ?? e.name}`;
  }
}, 4000);

const deadline = setInterval(() => {
  if (exited || Date.now() - started >= 20000) {
    clearInterval(deadline);
    const killedByProbe = !exited;
    if (!exited) {
      try { process.kill(-child.pid, "SIGTERM"); } catch {}
    }
    setTimeout(() => {
      console.log("=== captured stdout+stderr (verbatim) ===");
      console.log(out.trimEnd());
      console.log("=== probe result ===");
      console.log(`process exited on its own within 20 s: ${!killedByProbe}`);
      console.log(`exit info: ${JSON.stringify(exitInfo)}`);
      console.log(`killed by probe: ${killedByProbe}`);
      console.log(`request while running: ${probe}`);
      console.log(`contains "[Config] refusing to start": ${out.includes("[Config] refusing to start")}`);
      console.log(`contains "MissingEnvError": ${out.includes("MissingEnvError")}`);
      process.exit(0);
    }, 1500);
  }
}, 250);

==============================================================================
What Next 16.3.5 does with a throwing register() (measured, not assumed)
==============================================================================
- 'next start' prints Ready immediately. register() runs lazily, when the server is prepared for
  its first request, not at process start.
- When register() throws, Next logs '[Config] refusing to start: Missing required environment
  variable: SUPABASE_SERVICE_ROLE_KEY', then 'Failed to prepare server Error [MissingEnvError]'.
  The request is answered with 500 and the process does NOT exit (it was still running at 20 s).
- In the control, register() passes and the same request reaches the health route, which answers
  503 (its own database-unreachable status). The 500 in (b2) is therefore the boot check's.
- So 'refuses to start' means 'refuses to serve': every request fails, with one named log line that
  names the variable and never its value. That is what the must-have truth asks for (a server
  missing its env refuses to serve). Process exit is not something register() can force in this
  Next version. The unit tests are the authority on register()'s own behaviour.
