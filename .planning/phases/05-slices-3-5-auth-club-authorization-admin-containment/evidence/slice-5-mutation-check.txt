    #!/bin/zsh
    # Usage: cycles.sh — runs every mutation cycle, restoring each file from git.
    SP=/private/tmp/claude-501/-Users-adyan-Documents-GitHub-Event-Radar/d6f0f2c2-0ba5-4efb-b3ce-b5d843e8e77d/scratchpad
    cd /Users/adyan/Documents/GitHub/Event-Radar
    ADMIN="src/__tests__/api/admin/admin-guard-characterization.test.ts src/__tests__/api/admin/admin-guard-defect.test.ts"
    run_cycle() {
      local name="$1" keys="$2" files="$3" suites="$4" predict="$5"
      echo "--------------------------------------------------------------------------------"
      echo "CYCLE $name"
      echo "PREDICT: $predict"
      for k in ${=keys}; do echo "\$ python3 mutate.py $k"; python3 $SP/mutate.py $k; done
      echo "\$ git diff --stat -- $files"; git diff --stat -- ${=files}
      echo "\$ npx jest --ci --verbose $suites"
      npx jest --ci --verbose ${=suites} > $SP/cycle.out 2>&1; local rc=$?
      grep -E "✕" $SP/cycle.out | sed 's/ ([0-9]* ms)//'
      grep -E "^Tests:" $SP/cycle.out
      echo "jest exit=$rc"
      echo "\$ git checkout -- $files && git diff --exit-code -- $files"
      git checkout -- ${=files}; git diff --exit-code -- ${=files}; echo "restore exit=$?"
    }
    run_cycle 1 "1" "src/lib/admin.ts" "$ADMIN" \
      "admin helper admits everyone: every PRESERVE P1 student-403 row (32) goes red; D2 (batch student 401) goes red; D1 stays green (anonymous returns before the role read)"

      "admin users PATCH stops stripping admin: R1 (payload lacks admin) and R3 (self demotion) go red"
    run_cycle 5 "5" "src/app/users/[id]/page.tsx" "src/__tests__/pages/public-profile-defect.test.tsx" \
      "page calls notFound() for an anonymous viewer of a private target: P2 goes red; the public rows stay green"
