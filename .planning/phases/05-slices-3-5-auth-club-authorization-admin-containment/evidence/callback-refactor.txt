// Extract every top-level `  it(` block (from the `  it(` line to its closing `  });`)
// from the 05-04-head route.test.ts and the working tree, and compare byte for byte.
const { execSync } = require("child_process");
const fs = require("fs");
const FILE = "src/app/auth/callback/route.test.ts";
function blocks(src) {
  const lines = src.split("\n");
  const out = [];
  for (let i = 0; i < lines.length; i++) {
    if (lines[i].startsWith("  it(")) {
      let j = i;
      while (lines[j] !== "  });") j++;
      const text = lines.slice(i, j + 1).join("\n");
      const title = /it\("([^"]+)"/.exec(lines[i])[1];
      out.push({ title, text });
      i = j;
    }
  }
  return out;
}
